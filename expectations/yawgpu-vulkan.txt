# F-085: fragment builtins under Vulkan per-sample dispatch (@interpolate(...,sample) at
# sampleCount=4) — sample_mask input is the current sample's bit (Vulkan semantics) and
# position.xy is at the sample location, where the current CTS oracle expects the full
# coverage mask and the pixel center. NOT an implementation defect: yawgpu, wgpu-native and
# Dawn all behave this way on Vulkan (Dawn suppresses these cases, crbug.com/407144390);
# the WGSL WG resolved to respecify sample_mask to the Vulkan single-bit semantics
# (gpuweb/gpuweb#5457; CTS change gpuweb/cts#4510 pending). position is open in
# gpuweb/gpuweb#4777. Drop the sample_mask entries when cts#4510 merges and the port's
# oracle is updated; keep position until #4777 resolves.
webgpu:shader,execution,shader_io,fragment_builtins:inputs,sample_mask:nearFar="[0,1]";sampleCount=4;interpolation="linear,sample"
webgpu:shader,execution,shader_io,fragment_builtins:inputs,sample_mask:nearFar="[0,1]";sampleCount=4;interpolation="perspective,sample"
webgpu:shader,execution,shader_io,fragment_builtins:inputs,sample_mask:nearFar="[0.25,0.75]";sampleCount=4;interpolation="linear,sample"
webgpu:shader,execution,shader_io,fragment_builtins:inputs,sample_mask:nearFar="[0.25,0.75]";sampleCount=4;interpolation="perspective,sample"
webgpu:shader,execution,shader_io,fragment_builtins:inputs,position:nearFar="[0,1]";sampleCount=4;interpolation="linear,sample"
webgpu:shader,execution,shader_io,fragment_builtins:inputs,position:nearFar="[0,1]";sampleCount=4;interpolation="perspective,sample"
webgpu:shader,execution,shader_io,fragment_builtins:inputs,position:nearFar="[0.25,0.75]";sampleCount=4;interpolation="linear,sample"
webgpu:shader,execution,shader_io,fragment_builtins:inputs,position:nearFar="[0.25,0.75]";sampleCount=4;interpolation="perspective,sample"

# F-111: GPUExternalTexture is unsupported on the yawgpu Vulkan backend — naga's
# SPIR-V backend does not lower naga::ImageClass::External (the 3-plane + params
# expansion), so a render pipeline whose shader uses texture_external is rejected
# at SPIR-V codegen with a GPUInternalError ("external textures are not supported
# on the Vulkan backend"). The validation case expects pipeline creation to
# succeed, which Dawn does (it supports external textures on Vulkan). This is a
# documented feature gap, not a correctness defect (Metal has full external-texture
# support). Drop these entries if/when external-texture SPIR-V lowering lands.
webgpu:api,validation,render_pipeline,misc:external_texture:isAsync=false
webgpu:api,validation,render_pipeline,misc:external_texture:isAsync=true
