# F-085: sample_mask fragment builtin under Vulkan per-sample dispatch (@interpolate(...,sample)
# at sampleCount=4) — the sample_mask input is the current sample's single bit (Vulkan
# semantics) where the current CTS oracle expects the full coverage mask. NOT an implementation
# defect: yawgpu, wgpu-native and Dawn all behave this way on Vulkan (Dawn suppresses these
# cases, crbug.com/407144390); the WGSL WG resolved to respecify sample_mask to the Vulkan
# single-bit semantics (gpuweb/gpuweb#5457; CTS change gpuweb/cts#4510 pending). Drop these
# entries when cts#4510 merges and the port's oracle is updated.
#
# The sibling inputs,position cases were formerly xfail'd here too (position was thought to be
# equally spec-in-flux, gpuweb#4777), but the Dawn oracle now PASSES them by reconstructing the
# pixel center for @builtin(position) under sample-rate shading. yawgpu implements the same
# polyfill (see FINDINGS.md F-085), so position is a real pass and is no longer xfail'd here.
webgpu:shader,execution,shader_io,fragment_builtins:inputs,sample_mask:nearFar="[0,1]";sampleCount=4;interpolation="linear,sample"
webgpu:shader,execution,shader_io,fragment_builtins:inputs,sample_mask:nearFar="[0,1]";sampleCount=4;interpolation="perspective,sample"
webgpu:shader,execution,shader_io,fragment_builtins:inputs,sample_mask:nearFar="[0.25,0.75]";sampleCount=4;interpolation="linear,sample"
webgpu:shader,execution,shader_io,fragment_builtins:inputs,sample_mask:nearFar="[0.25,0.75]";sampleCount=4;interpolation="perspective,sample"

# F-111: external_texture render pipeline on yawgpu Vulkan — yawgpu deterministically rejects
# external textures on the Vulkan backend (Metal-only posture, mirroring wgpu), so this case
# (which expects pipeline creation to SUCCEED, as Dawn does) fails. Background: Tint's SPIR-V
# writer DOES lower `texture_external` (the old naga backend could not), but whether a driver
# accepts the multiplanar SPIR-V is GPU-DIVERGENT — native NVIDIA compiles it, Mesa rejects it,
# MoltenVK fails SPIR-V->MSL on the argument-buffer base type. Rather than ship that divergence,
# yawgpu detects the external-texture binding in select_{render,compute}_shader_source and returns
# a clean GPUInternalError BEFORE any SPIR-V reaches a driver (device-independent; commit 0985538).
# A real Vulkan re-home would require wiring the multiplanar plane/params bindings on the Vulkan
# HAL, not just relying on driver acceptance. Drop these entries if/when that lands.
# See specs/blocks/36-external-textures.md (R9) + memory f060-external-textures.
webgpu:api,validation,render_pipeline,misc:external_texture:isAsync=false
webgpu:api,validation,render_pipeline,misc:external_texture:isAsync=true

# CTS port-oracle quirk (not a yawgpu defect): draw,index_buffer_format_dirtying. A draw whose
# index-buffer format was invalidated by a pipeline change makes the render bundle an error bundle.
# The CTS oracle expects success, but EVERY backend rejects it -- including the Dawn reference impl,
# which fails the same 2 cases identically (Dawn-confirmed 2026-06-28). A port-oracle issue, not a
# backend defect; also listed in expectations/yawgpu.txt (Metal). See docs/FINDINGS.md.
webgpu:api,validation,encoding,cmds,render,draw:index_buffer_format_dirtying:*
